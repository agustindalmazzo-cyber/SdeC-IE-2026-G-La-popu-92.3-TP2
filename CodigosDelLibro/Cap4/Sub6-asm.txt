; subroutina _calc_sum
; halla la suma de los enteros de 1 hasta n
; Par´ametros:
; n - hasta d´onde se suma (en [ebp + 8])
; Valor de retorno:
; la suma
; pseudoc´odigo en C:
; int calc_sum( int n )
; {
; int i, sum = 0;
; for( i=1; i <= n; i++ )
; sum += i;
; return sum;
; }
segment .text
global _calc_sum
;
; local variable:
; sum at [ebp-4]
_calc_sum:
enter 4,0 ; hace espacio en la pila para sum

mov dword [ebp-4],0 ; sum = 0
mov ecx, 1 ; ecx es i en el pseudoc´odigo
for_loop:
cmp ecx, [ebp+8] ; cmp i and n
jnle end_for ; si no i <= n, sale

add [ebp-4], ecx ; sum += i
inc ecx
jmp short for_loop

end_for:
mov eax, [ebp-4] ; eax = sum

leave
ret
