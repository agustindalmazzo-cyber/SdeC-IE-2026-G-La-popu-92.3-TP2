; subroutinea _calc_sum
; halla la suma de los enteros de 1 hasta n
; Parametros:
; n - hasta d´onde sumar (en [ebp + 8])
; sump - apuntador a un int para almacenar sum (en [ebp + 12])
; pseudoc´odigo en C:
; void calc_sum( int n, int * sump )
; {
; int i, sum = 0;
; for( i=1; i <= n; i++ )
; sum += i;
; *sump = sum;
; }

segment .text
global _calc_sum
;
; local variable:
; sum at [ebp-4]
_calc_sum:
enter 4,0 ; hace espacio para sum en la pila
push ebx ; IMPORTANTE!

mov dword [ebp-4],0 ; sum = 0
dump_stack 1, 2, 4 ; imprime la pila desde ebp-8 hasta ebp+16
mov ecx, 1 ; ecx es i en el pseudoc´odigo
for_loop:
cmp ecx, [ebp+8] ; cmp i y n
jnle end_for ; si no i <= n, sale

add [ebp-4], ecx ; sum += i
inc ecx
jmp short for_loop

end_for:
mov ebx, [ebp+12] ; ebx = sump
mov eax, [ebp-4] ; eax = sum
mov [ebx], eax

pop ebx ; restaura ebx
leave
ret
