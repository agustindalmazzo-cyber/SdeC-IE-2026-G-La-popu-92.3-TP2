%include "asm_io.inc"

segment .data
sum dd 0

segment .bss
input resd 1

segment .text
global _asm_main
extern get_int, print_sum
 _asm_main:
enter 0,0 ; setup routine
pusha

mov edx, 1 ; edx es ’i’ en el pseudoc´odigo
while_loop:
push edx ; guarda i en la pila
push dword input ; empuja la direcci´on input en la pila
call get_int
add esp, 8 ; quita i e &input de la pila

mov eax, [input]
cmp eax, 0
je end_while

add [sum], eax ; sum += input

inc edx
jmp short while_loop

end_while:
push dword [sum] ; empuja el valor de sum de la pila
call print_sum
pop ecx ; quita sum de la pila
popa
leave
ret
